# Native test suite for flutter_soloud: one target and one ctest test per
# executable built by the scripts in test/common/.
#
#   cmake -S test -B build/native_tests -DCMAKE_BUILD_TYPE=RelWithDebInfo
#   cmake --build build/native_tests --parallel
#   ctest --test-dir build/native_tests --output-on-failure -j4
#
# Options:
#   -DSOLOUD_TEST_SANITIZE=address   AddressSanitizer + UBSan (MSVC: /fsanitize=address)
#   -DSOLOUD_TEST_SANITIZE=thread    ThreadSanitizer (not MSVC)
#   -DSOLOUD_TEST_COVERAGE=ON        clang/gcc coverage instrumentation (see test/README.md)
#   -DXIPH_PREFIX=<dir>              where libogg/libvorbis/libopus/libFLAC are installed
#
# The part below the marker is generated by test/support/gen_cmake.py from
# the bash scripts, which remain the reference for each test. Edit this
# header in test/support/CMakeLists.header.cmake, then rerun the generator.

cmake_minimum_required(VERSION 3.16)
project(flutter_soloud_native_tests LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_EXTENSIONS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)

find_package(Threads REQUIRED)
enable_testing()

if(MSVC)
  add_compile_options(/W0 /bigobj /utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  add_compile_options(-w)
endif()

set(SOLOUD_TEST_SANITIZE "" CACHE STRING "address, thread or empty")
if(SOLOUD_TEST_SANITIZE STREQUAL "address")
  if(MSVC)
    add_compile_options(/fsanitize=address)
  else()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    # Leaks inside system libraries, not plugin code (see lsan.supp).
    set(SOLOUD_TEST_ENV
      "LSAN_OPTIONS=suppressions=${REPO_ROOT}/test/support/lsan.supp:print_suppressions=0")
  endif()
elseif(SOLOUD_TEST_SANITIZE STREQUAL "thread")
  add_compile_options(-fsanitize=thread)
  add_link_options(-fsanitize=thread)
endif()

option(SOLOUD_TEST_COVERAGE "Instrument for coverage" OFF)
if(SOLOUD_TEST_COVERAGE AND NOT MSVC)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    add_compile_options(-fprofile-instr-generate -fcoverage-mapping)
    add_link_options(-fprofile-instr-generate)
  else()
    add_compile_options(--coverage)
    add_link_options(--coverage)
  endif()
endif()

# Per-platform replacements for the Apple-only sources the scripts use on macOS.
set(SOLOUD_APPLE_ONLY_SOURCES "")
set(SOLOUD_PLATFORM_LIBS "")
set(SOLOUD_M_LIB m)
if(APPLE)
  set(SOLOUD_OS_DECODER_SOURCE "${REPO_ROOT}/src/native_decoder/os_decoder_apple.mm")
  set(SOLOUD_AAC_DECODER_SOURCE "${REPO_ROOT}/src/audiobuffer/aac_stream_decoder_apple.mm")
  set(SOLOUD_APPLE_ONLY_SOURCES "${REPO_ROOT}/src/soloud_miniaudio_objc.mm")
  set(SOLOUD_PLATFORM_LIBS
    "-framework Foundation" "-framework CoreFoundation" "-framework CoreAudio"
    "-framework AudioToolbox" "-framework AVFoundation")
elseif(WIN32)
  set(SOLOUD_OS_DECODER_SOURCE "${REPO_ROOT}/src/native_decoder/os_decoder_windows.cpp")
  set(SOLOUD_AAC_DECODER_SOURCE "${REPO_ROOT}/src/audiobuffer/aac_stream_decoder_windows.cpp")
  set(SOLOUD_PLATFORM_LIBS mfplat mfreadwrite mfuuid ole32 propsys shlwapi winmm)
  set(SOLOUD_M_LIB "")
else()
  set(SOLOUD_OS_DECODER_SOURCE "${REPO_ROOT}/src/native_decoder/os_decoder_linux.cpp")
  set(SOLOUD_AAC_DECODER_SOURCE "${REPO_ROOT}/src/audiobuffer/aac_stream_decoder_linux.cpp")
  set(SOLOUD_PLATFORM_LIBS ${CMAKE_DL_LIBS})
endif()

# Xiph libraries (tests that need them are skipped when they are missing).
if(NOT XIPH_PREFIX)
  if(DEFINED ENV{XIPH_PREFIX})
    set(XIPH_PREFIX "$ENV{XIPH_PREFIX}")
  elseif(APPLE AND EXISTS "/opt/homebrew")
    set(XIPH_PREFIX "/opt/homebrew")
  else()
    set(XIPH_PREFIX "/usr/local")
  endif()
endif()
find_path(XIPH_INCLUDE_DIR ogg/ogg.h HINTS "${XIPH_PREFIX}/include")
set(SOLOUD_XIPH_FOUND ON)
foreach(lib ogg vorbis vorbisenc vorbisfile opus FLAC)
  string(TOUPPER ${lib} LIBU)
  find_library(XIPH_${LIBU}_LIB NAMES ${lib} lib${lib} HINTS "${XIPH_PREFIX}/lib")
  if(NOT XIPH_${LIBU}_LIB)
    set(SOLOUD_XIPH_FOUND OFF)
  endif()
endforeach()
if(NOT XIPH_INCLUDE_DIR)
  set(SOLOUD_XIPH_FOUND OFF)
endif()
if(NOT SOLOUD_XIPH_FOUND)
  message(STATUS "Xiph libraries not found under ${XIPH_PREFIX}: tests that need them are not built")
endif()
